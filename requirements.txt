# Runtime dependencies — exactly what `api/`, `pyref/`, and `sim/` import.
# This file IS the Docker runtime layer, so keep it minimal and pinned.
#
# Deliberately absent: osmnx and networkx. They are ingestion-only (see
# requirements-ingest.txt) and pull in geopandas/shapely/pyproj/pyogrio/
# pandas, which roughly quadruples the image. Nothing served at request
# time imports them.
#
# Also absent: pybind11 and setuptools. `core/pyproject.toml` declares them
# as build-system requires, so pip's build isolation supplies them during
# `pip install ./core` — they are never needed at runtime.

httpx==0.28.1
numpy==2.5.3
pydantic==2.13.5
scipy==1.18.1
uvicorn[standard]==0.51.0

# Imported lazily, and only when SR_REDIS_URL / [api].redis_url is set —
# api/ratelimit.py falls back to an in-process token bucket otherwise, so local
# dev and the test suite never import this. It is pinned here anyway because
# the image is what gets scaled out, and a multi-replica deployment that has to
# rebuild the image to get a shared rate limiter would just skip the limiter.
redis==8.1.0

tzdata==2026.3
